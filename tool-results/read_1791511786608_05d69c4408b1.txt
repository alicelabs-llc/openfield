     1→//! On-disk state: the ledger, search history, config, and the crash marker.
     2→//!
     3→//! Three rules, all of them about not losing a user's data:
     4→//!
     5→//! 1. **Every write is atomic** — temp file in the same directory, then rename.
     6→//!    A crash mid-write leaves either the old file or the new one, never half of
     7→//!    one (`FR-55`). Same directory matters: a rename across filesystems is a
     8→//!    copy, and stops being atomic.
     9→//! 2. **A corrupt file is quarantined, never deleted and never overwritten.**
    10→//!    We rename it to `.corrupt`, start from defaults, and say so (`FR-54`).
    11→//!    The user's file may be the only copy of something they care about.
    12→//! 3. **One bad row does not lose the file.** A ledger with nineteen good
    13→//!    entries and one malformed one restores nineteen items.
    14→//!
    15→//! Everything is addressed through [`Store`] rather than reading the environment
    16→//! at each call, so tests run against a temp directory in parallel without
    17→//! fighting over `HARBOUR_STATE_DIR`.
    18→//!
    19→//! `dead_code` is allowed module-wide until the app loop calls [`Store`] on the
    20→//! boot and quit paths (E2/E3). Every item here is covered by tests; the lint
    21→//! only means nothing in `main` reaches it yet. Remove the allow as the wiring
    22→//! lands.
    23→

   123→/// Outcome of a load that can degrade. The caller turns `Recovered` into the
   124→/// banner the user sees — a silent fallback is how people lose data without
   125→/// noticing.
   126→#[derive(Debug, Clone, PartialEq)]
   127→pub enum Loaded<T> {
   128→    /// Loaded cleanly, or the file simply did not exist yet.
   129→    Ok(T),
   130→    /// Something was wrong. `value` is the best we could salvage.
   131→    Recovered { value: T, warning: String },
   132→}
   133→
   134→impl<T> Loaded<T> {
   135→    pub fn value(self) -> T {
   136→        match self {
   137→            Loaded::Ok(v) => v,
   138→            Loaded::Recovered { value, .. } => value,
   139→        }
   140→    }
   141→
   142→    pub fn warning(&self) -> Option<&str> {
   143→        match self {
   144→            Loaded::Ok(_) => None,
   145→            Loaded::Recovered { warning, .. } => Some(warning),
   146→        }
   147→    }
   148→}

   186→    // --- ledger -------------------------------------------------------------
   187→
   188→    /// Writes the ledger atomically. Only durable fields are in [`QueueItem`],
   189→    /// so this is called on status transitions rather than on every poll tick.
   190→    pub fn save_ledger(&self, items: &[QueueItem]) -> io::Result<()> {
   191→        let json = serde_json::to_vec_pretty(items)
   192→            .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
   193→        atomic_write(&self.ledger_path(), &json)
   194→    }
   195→
   196→    /// Loads the ledger, salvaging as much as possible.
   197→    ///
   198→    /// A missing file is a clean empty start. A file that will not parse at all
   199→    /// is quarantined. A file that parses as an array keeps every row that is a
   200→    /// valid item and reports how many were dropped.
   201→    pub fn load_ledger(&self) -> Loaded<Vec<QueueItem>> {
   202→        let path = self.ledger_path();
   203→        let raw = match fs::read_to_string(&path) {
   204→            Ok(raw) => raw,
   205→            Err(e) if e.kind() == io::ErrorKind::NotFound => return Loaded::Ok(Vec::new()),
   206→            Err(e) => {
   207→                return Loaded::Recovered {
   208→                    value: Vec::new(),
   209→                    warning: format!("could not read {}: {e}", path.display()),
   210→                };
   211→            }
   212→        };
   213→
   214→        // Parse to generic JSON first so one malformed entry costs one entry
   215→        // rather than the whole file.
   216→        let rows: Vec<serde_json::Value> = match serde_json::from_str(&raw) {
   217→            Ok(serde_json::Value::Array(rows)) => rows,
   218→            _ => {
   219→                let warning = match quarantine(&path) {
   220→                    Ok(moved) => format!(
   221→                        "{} was unreadable; kept a copy at {} and started with an empty queue",
   222→                        path.display(),
   223→                        moved.display()
   224→                    ),
   225→                    Err(e) => format!(
   226→                        "{} was unreadable and could not be set aside ({e}); \
   227→                         started with an empty queue",
   228→                        path.display()
   229→                    ),
   230→                };
   231→                return Loaded::Recovered {
   232→                    value: Vec::new(),
   233→                    warning,
   234→                };
   235→            }
   236→        };
   237→
   238→        let total = rows.len();
   239→        let items: Vec<QueueItem> = rows
   240→            .into_iter()
   241→            .filter_map(|row| serde_json::from_value(row).ok())
   242→            .collect();
   243→
   244→        let dropped = total - items.len();
   245→        if dropped == 0 {
   246→            Loaded::Ok(items)
   247→        } else {
   248→            Loaded::Recovered {
   249→                warning: format!(
   250→                    "{dropped} of {total} saved downloads could not be read and were skipped"
   251→                ),
   252→                value: items,
   253→            }
   254→        }
   255→    }

   257→    // --- search history -----------------------------------------------------
   258→
   259→    /// Saves search queries, newest first, capped at [`HISTORY_CAP`].
   260→    pub fn save_history(&self, queries: &[String]) -> io::Result<()> {
   261→        let capped: Vec<&String> = queries.iter().take(HISTORY_CAP).collect();
   262→        let json = serde_json::to_vec_pretty(&capped)
   263→            .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))?;
   264→        atomic_write(&self.history_path(), &json)
   265→    }
   266→
   267→    /// Loads search history. Corrupt history is worth nothing, so it degrades
   268→    /// to empty without quarantine noise.
   269→    pub fn load_history(&self) -> Loaded<Vec<String>> {
   270→        let path = self.history_path();
   271→        match fs::read_to_string(&path) {
   272→            Err(e) if e.kind() == io::ErrorKind::NotFound => Loaded::Ok(Vec::new()),
   273→            Err(e) => Loaded::Recovered {
   274→                value: Vec::new(),
   275→                warning: format!("could not read search history: {e}"),
   276→            },
   277→            Ok(raw) => match serde_json::from_str::<Vec<String>>(&raw) {
   278→                Ok(mut v) => {
   279→                    v.truncate(HISTORY_CAP);
   280→                    Loaded::Ok(v)
   281→                }
   282→                Err(_) => Loaded::Recovered {
   283→                    value: Vec::new(),
   284→                    warning: "search history was unreadable and has been reset".into(),
   285→                },
   286→            },
   287→        }
   288→    }
   289→
   290→    /// Records a query at the front, de-duplicated, capped.
   291→    pub fn push_history(&self, existing: &mut Vec<String>, query: &str) -> io::Result<()> {
   292→        let query = query.trim();
   293→        if query.is_empty() {
   294→            return Ok(());
   295→        }
   296→        existing.retain(|q| q != query);
   297→        existing.insert(0, query.to_owned());
   298→        existing.truncate(HISTORY_CAP);
   299→        self.save_history(existing)
   300→    }

   310→    /// Loads config, falling back loudly.
   311→    ///
   312→    /// A missing config is a first run, not an error. An unparseable one keeps
   313→    /// defaults **and leaves the file alone** — overwriting someone's config
   314→    /// because we could not read it is not a recovery, it is a second failure.
   315→    pub fn load_config(&self) -> Loaded<Config> {
   316→        let path = self.config_path();
   317→        match fs::read_to_string(&path) {
   318→            Err(e) if e.kind() == io::ErrorKind::NotFound => Loaded::Ok(Config::default()),
   319→            Err(e) => Loaded::Recovered {
   320→                value: Config::default(),
   321→                warning: format!("could not read {}: {e}", path.display()),
   322→            },
   323→            Ok(raw) => match toml::from_str::<Config>(&raw) {
   324→                Ok(cfg) => Loaded::Ok(cfg),
   325→                Err(e) => Loaded::Recovered {
   326→                    value: Config::default(),
   327→                    warning: format!(
   328→                        "{} is not valid TOML ({}); using defaults. Your file has been left \
   329→                         untouched.",
   330→                        path.display(),
   331→                        e.message()
   332→                    ),
   333→                },
   334→            },
   335→        }
   336→    }

   345→    /// Writes the crash marker. Called immediately before state is handed to the
   346→    /// engine.
   347→    pub fn arm_boot_marker(&self) -> io::Result<()> {
   348→        ensure_parent(&self.boot_marker_path())?;
   349→        fs::write(self.boot_marker_path(), b"harbour boot in progress\n")
   350→    }
   351→
   352→    /// Clears the crash marker. **Flush the ledger before calling this**: a
   353→    /// crash between the two would otherwise leave a clean marker over stale
   354→    /// state, and bootguard would stand down exactly when it was needed.
   355→    pub fn disarm_boot_marker(&self) -> io::Result<()> {
   356→        match fs::remove_file(self.boot_marker_path()) {
   357→            Ok(()) => Ok(()),
   358→            Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(()),
   359→            Err(e) => Err(e),
   360→        }
   361→    }
   362→
   363→    /// The exit path, in the one order that is safe: flush state, *then* stand
   364→    /// the crash breaker down.
   365→    pub fn flush_and_disarm(&self, items: &[QueueItem]) -> io::Result<()> {
   366→        self.save_ledger(items)?;
   367→        self.disarm_boot_marker()
   368→    }

   370→    // --- cache (FR-80) ------------------------------------------------------
   371→
   372→    /// Wipes `cache/search/` plus unused `.torrent` files and watch-now scratch
   373→    /// dirs whose infohash is not in `keep`. Never touches the download
   374→    /// directory or the ledger — leftover pieces there are `shift+X` (#77).
   375→    pub fn clear_cache(&self, keep: &HashSet<String>) -> io::Result<u64> {
   376→        let mut freed = wipe_tree(&paths::search_cache_root(&self.root))?;
   377→        freed += purge_unused_torrents(&self.root, keep)?;
   378→        freed += purge_unused_watch_dirs(&self.root, keep)?;
   379→        Ok(freed)
   380→    }

   382→    /// Deletes `<state>/cache/<hash>/` watch-now scratch for one torrent.
   383→    /// Missing or invalid hashes are success: forget must stay idempotent.
   384→    pub fn remove_watch_cache(&self, id: &str) -> io::Result<()> {
   385→        let Some(dir) = paths::watch_cache_dir(&self.root, id) else {
   386→            return Ok(());
   387→        };
   388→        match fs::remove_dir_all(&dir) {
   389→            Ok(()) => Ok(()),
   390→            Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(()),
   391→            Err(e) => Err(e),
   392→        }
   393→    }
   394→}

   396→/// Writes `bytes` to `path` atomically.
   397→///
   398→/// The temp file is a sibling so the rename stays within one filesystem, and it
   399→/// is removed on a failed write so a crashed run does not leave litter that
   400→/// looks like state.
   401→pub fn atomic_write(path: &Path, bytes: &[u8]) -> io::Result<()> {
   402→    ensure_parent(path)?;
   403→    let tmp = path.with_extension(format!(
   404→        "{}.tmp",
   405→        path.extension().and_then(|e| e.to_str()).unwrap_or("new")
   406→    ));
   407→    if let Err(e) = fs::write(&tmp, bytes) {
   408→        let _ = fs::remove_file(&tmp);
   409→        return Err(e);
   410→    }
   411→    // fsync before rename (FR-55): without it a power loss can make the rename
   412→    // durable while the data blocks are not, leaving a zero-length or partial
   413→    // ledger — the exact "no partial file" promise this function makes. The
   414→    // handle must be opened write-enabled: sync_all on Windows requires it.
   415→    if let Err(e) = fs::OpenOptions::new()
   416→        .write(true)
   417→        .open(&tmp)
   418→        .and_then(|f| f.sync_all())
   419→    {
   420→        let _ = fs::remove_file(&tmp);
   421→        return Err(e);
   422→    }
   423→    // rename replaces the destination on both Unix and Windows.
   424→    if let Err(e) = fs::rename(&tmp, path) {
   425→        let _ = fs::remove_file(&tmp);
   426→        return Err(e);
   427→    }
   428→    Ok(())
   429→}
   430→
   431→fn ensure_parent(path: &Path) -> io::Result<()> {
   432→    match path.parent() {
   433→        Some(dir) if !dir.as_os_str().is_empty() => fs::create_dir_all(dir),
   434→        _ => Ok(()),
   435→    }
   436→}

   438→/// Moves an unreadable file aside, never deleting it.
   439→fn quarantine(path: &Path) -> io::Result<PathBuf> {
   440→    let target = path.with_extension(format!(
   441→        "{}.corrupt",
   442→        path.extension().and_then(|e| e.to_str()).unwrap_or("bak")
   443→    ));
   444→    fs::rename(path, &target)?;
   445→    Ok(target)
   446→}

   448→/// Recursively deletes `dir` and returns how many bytes it held. Missing is 0.
   449→fn wipe_tree(dir: &Path) -> io::Result<u64> {
   450→    if !dir.exists() {
   451→        return Ok(0);
   452→    }
   453→    let bytes = dir_size(dir);
   454→    fs::remove_dir_all(dir)?;
   455→    Ok(bytes)
   456→}
   457→
   458→fn dir_size(dir: &Path) -> u64 {
   459→    let Ok(entries) = fs::read_dir(dir) else {
   460→        return 0;
   461→    };
   462→    entries
   463→        .flatten()
   464→        .map(|entry| {
   465→            let path = entry.path();
   466→            if path.is_dir() {
   467→                dir_size(&path)
   468→            } else {
   469→                entry.metadata().map(|m| m.len()).unwrap_or(0)
   470→            }
   471→        })
   472→        .sum()
   473→}
   474→
   475→fn purge_unused_torrents(root: &Path, keep: &HashSet<String>) -> io::Result<u64> {
   476→    let dir = paths::torrent_cache_dir(root);
   477→    let Ok(entries) = fs::read_dir(&dir) else {
   478→        return Ok(0);
   479→    };
   480→    let mut freed = 0_u64;
   481→    for entry in entries.flatten() {
   482→        let path = entry.path();
   483→        if path.extension().and_then(|e| e.to_str()) != Some("torrent") {
   484→            continue;
   485→        }
   486→        let Some(stem) = path.file_stem().and_then(|s| s.to_str()) else {
   487→            continue;
   488→        };
   489→        if keep.contains(stem) {
   490→            continue;
   491→        }
   492→        let size = entry.metadata().map(|m| m.len()).unwrap_or(0);
   493→        match fs::remove_file(&path) {
   494→            Ok(()) => freed += size,
   495→            Err(e) if e.kind() == io::ErrorKind::NotFound => {}
   496→            Err(e) => return Err(e),
   497→        }
   498→    }
   499→    Ok(freed)
   500→}
   501→
   502→fn purge_unused_watch_dirs(root: &Path, keep: &HashSet<String>) -> io::Result<u64> {
   503→    let dir = paths::cache_dir(root);
   504→    let Ok(entries) = fs::read_dir(&dir) else {
   505→        return Ok(0);
   506→    };
   507→    let mut freed = 0_u64;
   508→    for entry in entries.flatten() {
   509→        let path = entry.path();
   510→        if !path.is_dir() {
   511→            continue;
   512→        }
   513→        let Some(name) = entry.file_name().to_str().map(str::to_owned) else {
   514→            continue;
   515→        };
   516→        // Only hash-keyed watch scratch (`cache/<hash>/`), never torrents/search.
   517→        if paths::watch_cache_dir(root, &name).as_deref() != Some(path.as_path()) {
   518→            continue;
   519→        }
   520→        if keep.contains(&name) {
   521→            continue;
   522→        }
   523→        let size = dir_size(&path);
   524→        match fs::remove_dir_all(&path) {
   525→            Ok(()) => freed += size,
   526→            Err(e) if e.kind() == io::ErrorKind::NotFound => {}
   527→            Err(e) => return Err(e),
   528→        }
   529→    }
   530→    Ok(freed)
   531→}

   533→/// Binary units for the clear-cache banner, matching the downloads view.
   534→pub(crate) fn format_bytes(bytes: u64) -> String {
   535→    const UNITS: &[&str] = &["B", "KiB", "MiB", "GiB", "TiB"];
   536→    let mut value = bytes as f64;
   537→    let mut unit = 0;
   538→    while value >= 1024.0 && unit < UNITS.len() - 1 {
   539→        value /= 1024.0;
   540→        unit += 1;
   541→    }
   542→    if unit == 0 {
   543→        format!("{bytes} B")
   544→    } else {
   545→        format!("{value:.1} {}", UNITS[unit])
   546→    }
   547→}

   555→    /// Per-test temp directory. No `tempfile` dependency for what is six lines;
   556→    /// the counter keeps parallel tests from colliding.
   557→    fn temp_store(tag: &str) -> Store {
   558→        static N: AtomicU32 = AtomicU32::new(0);
   559→        let n = N.fetch_add(1, Ordering::Relaxed);
   560→        let dir =
   561→            std::env::temp_dir().join(format!("harbour-test-{tag}-{}-{n}", std::process::id()));
   562→        let _ = fs::remove_dir_all(&dir);
   563→        fs::create_dir_all(&dir).expect("create temp dir");
   564→        Store::new(dir)
   565→    }
