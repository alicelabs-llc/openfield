   154→def _get_cached(module: str, target: str) -> Optional[Dict]:
   155→    path = _cache_key(module, target)
   156→    if not os.path.exists(path):
   157→        return None
   158→    try:
   159→        age = time.time() - os.path.getmtime(path)
   160→        if age > _CACHE_TTL:
   161→            os.remove(path)
   162→            return None
   163→        with open(path, "r", encoding="utf-8") as f:
   164→            return json.load(f)
   165→    except Exception:
   166→        return None

   175→def _geocode_place(query: str) -> Optional[Dict]:
   176→    import hashlib
   177→    cache_path = os.path.join(_CACHE_DIR, "geocode_" + hashlib.md5(query.lower().encode()).hexdigest() + ".json")
   178→    try:
   179→        if os.path.exists(cache_path) and time.time() - os.path.getmtime(cache_path) < 30 * 86400:
   180→            with open(cache_path, "r", encoding="utf-8") as f:
   181→                return json.load(f)
   182→    except Exception:
   183→        pass
   184→    place = None
   185→    try:
   186→        r = _requests.get(
   187→            "https://nominatim.openstreetmap.org/search",
   188→            params={"q": query, "format": "json", "limit": 1},
   189→            headers={"User-Agent": USER_AGENT},
   190→            timeout=8,
   191→        )
   192→        arr = r.json()
   193→        if arr:
   194→            bb = arr[0].get("boundingbox")
   195→            bbox = [float(bb[0]), float(bb[1]), float(bb[2]), float(bb[3])] if bb and len(bb) == 4 else None
   196→            place = {"lat": float(arr[0]["lat"]), "lng": float(arr[0]["lon"]), "bbox": bbox}
   197→    except Exception:
   198→        return None
   199→    try:
   200→        with open(cache_path, "w", encoding="utf-8") as f:
   201→            json.dump(place, f)

   283→def _is_public_ip(addr) -> bool:
   284→    return not (
   285→        addr.is_private or addr.is_loopback or addr.is_reserved
   286→        or addr.is_link_local or addr.is_multicast or addr.is_unspecified
   287→    )

   289→def _resolve_all_public(hostname: str) -> str:
   290→    import ipaddress
   291→    import socket
   292→    try:
   293→        infos = socket.getaddrinfo(hostname, None)
   294→    except socket.gaierror:
   295→        try:
   296→            fallback_ip = socket.gethostbyname(hostname)
   297→        except socket.gaierror:
   298→            raise ValueError("webhook_url hostname cannot be resolved")
   299→        try:
   300→            addr = ipaddress.ip_address(fallback_ip)
   301→        except ValueError:
   302→            raise ValueError("webhook_url resolved to an invalid address")
   303→        if not _is_public_ip(addr):
   304→            raise ValueError("webhook_url resolves to a private/internal address")
   305→        return str(addr)
   306→    seen = set()
   307→    first_public = None
   308→    for fam, _t, _p, _c, sa in infos:
   309→        ip_str = sa[0]
   310→        if ip_str in seen:
   311→            continue
   312→        seen.add(ip_str)
   313→        try:
   314→            addr = ipaddress.ip_address(ip_str)
   315→        except ValueError:
   316→            raise ValueError("webhook_url resolved to an invalid address")
   317→        if not _is_public_ip(addr):
   318→            raise ValueError("webhook_url resolves to a private/internal address")
   319→        if first_public is None:
   320→            first_public = str(addr)
   321→    if not seen:
   322→        raise ValueError("webhook_url hostname did not resolve")
   323→    return first_public

   326→def _validate_webhook_url(url: str) -> str:
   327→    from urllib.parse import urlparse
   328→    if not url or len(url) > 2048:
   329→        raise ValueError("webhook_url is empty or too long")
   330→    parsed = urlparse(url)
   331→    if parsed.scheme not in ("http", "https") or not parsed.hostname:
   332→        raise ValueError("webhook_url must be http(s) with a hostname")
   333→    _resolve_all_public(parsed.hostname)
   334→    return url

   373→def _start_watchlist_scheduler() -> None:
   374→    if not env_flag("WATCHLIST_SCHEDULER", True):
   375→        return
   376→    # Look for the running thread rather than a module-level flag. Re-importing
   377→    # this module (the test suite does it on every _load_app) re-runs every
   378→    # top-level assignment, so a flag came back False and the next startup
   379→    # launched another scheduler thread alongside the one still looping.
   380→    if any(t.name == _WATCHLIST_THREAD_NAME and t.is_alive() for t in threading.enumerate()):
   381→        return
   382→    threading.Thread(
   383→        target=_watchlist_scheduler_loop, name=_WATCHLIST_THREAD_NAME, daemon=True
   384→    ).start()

   386→def _send_webhook(url: str, payload: Dict[str, Any]) -> None:
   387→    from urllib.parse import urlparse
   388→    try:
   389→        parsed = urlparse(url)
   390→        original_url = httpx.URL(url)
   391→        if parsed.scheme not in ("http", "https") or not parsed.hostname:
   392→            return
   393→        pinned_ip = _resolve_all_public(parsed.hostname)
   394→    except Exception:
   395→        return
   396→
   397→    webhook_format = os.environ.get("WEBHOOK_FORMAT", "raw")
   398→    if webhook_format == "slack":
   399→        payload = format_slack(payload)
   400→    elif webhook_format == "discord":
   401→        payload = format_discord(payload)
   402→
   403→    headers = {"Content-Type": "application/json", "User-Agent": USER_AGENT}
   404→    if WEBHOOK_SECRET:
   405→        headers["X-Prism-Secret"] = WEBHOOK_SECRET
   406→    original_host = original_url.raw_host.decode("ascii")
   407→    host_header = f"[{original_host}]" if ":" in original_host else original_host
   408→    if original_url.port is not None:
   409→        host_header = f"{host_header}:{original_url.port}"
   410→    headers["Host"] = host_header
   411→    try:
   412→        with httpx.Client(
   413→            timeout=10, follow_redirects=False, trust_env=False, verify=True,
   414→        ) as client:
   415→            request = client.build_request(
   416→                "POST",
   417→                original_url.copy_with(host=pinned_ip),
   418→                json=payload,
   419→                headers=headers,
   420→                extensions={"sni_hostname": original_host},
   421→            )
   422→            client.send(request, follow_redirects=False)
   423→    except Exception:
   424→        pass

   426→def _detect_type(target: str) -> str:
   427→    if "@" in target:
   428→        return "email"
   429→    parts = target.replace("+", "").replace("-", "").replace(" ", "")
   430→    if parts.isdigit():
   431→        return "phone"
   432→    t = target.lstrip("@")
   433→    if t.startswith("t.me/") or t.startswith("telegram.me/"):
   434→        return "telegram"
   435→    if "." in target:
   436→        segs = target.split(".")
   437→        if len(segs) == 4 and all(s.isdigit() for s in segs):
   438→            return "ip"
   439→        return "domain"
   440→    return "username"

   442→async def _push(scan_id: str, msg: Dict) -> None:
   443→    scan = _scans.get(scan_id)
   444→    if scan is not None:
   445→        if "progress" not in scan:
   446→            scan["progress"] = []
   447→        scan["progress"].append(msg)
   448→        _save_scan(scan_id, scan)
   449→    q = _queues.get(scan_id)
   450→    if q:
   451→        await q.put(msg)

   453→_CACHED_MODULES = {"shodan", "hlr", "virustotal", "abuseipdb", "geoip"}

   455→def _done_message(name: str, result: Any, cached: bool = False) -> Dict[str, Any]:
   456→    status = classify(result)
   457→    msg: Dict[str, Any] = {"type": "module_done", "module": name, "status": status}
   458→    if cached:
   459→        msg["cached"] = True
   460→    reason = reason_for(result)
   461→    if reason and status != OK:
   462→        msg["reason"] = reason
   463→        if status == ERROR:
   464→            msg["error"] = reason
   465→    return msg

   468→async def _run_module(scan_id: str, name: str, coro_or_func, *args, **kwargs) -> Any:
   469→    await _push(scan_id, {"type": "module_start", "module": name})
   470→    cache_target = args[0] if args else None
   471→    force_refresh = bool(_scans.get(scan_id, {}).get("force_refresh"))
   472→    if not force_refresh and name in _CACHED_MODULES and cache_target:
   473→        cached = _get_cached(name, str(cache_target))
   474→        # Ignore legacy non-OK cache entries so the module can re-run.
   475→        if cached is not None and classify(cached) == OK:
   476→            await _push(scan_id, _done_message(name, cached, cached=True))
   477→            return cached
   478→    try:
   479→        if asyncio.iscoroutinefunction(coro_or_func):
   480→            result = await coro_or_func(*args, **kwargs)
   481→        else:
   482→            loop = asyncio.get_running_loop()
   483→            result = await loop.run_in_executor(None, lambda: coro_or_func(*args, **kwargs))
   484→        status = classify(result)
   485→        if isinstance(result, dict) and "status" not in result:
   486→            result["status"] = status
   487→        # Cache only successful results so a missing key is not frozen in cache.
   488→        if name in _CACHED_MODULES and cache_target and status == OK:
   489→            _set_cache(name, str(cache_target), result)
   490→        await _push(scan_id, _done_message(name, result))
   491→        return result
   492→    except Exception as exc:
   493→        await _push(scan_id, {"type": "module_done", "module": name, "status": ERROR, "error": str(exc)})
   494→        return {"error": str(exc), "status": ERROR}

   496→async def _execute_scan(scan_id: str, target: str, scan_type: str, modules: list, webhook_url: Optional[str] = None) -> None:
   497→    results: Dict[str, Any] = {}
   498→    all_modules = not modules
   499→
   500→    def want(name: str) -> bool:
   501→        return all_modules or name in modules
   502→
   503→    try:
   504→        if scan_type in ("domain", "ip"):

  1533→def _is_reserved_frontend_path(path: str) -> bool:

  1537→def _safe_frontend_file(path: str) -> Optional[Path]:
  1538→    if not _FRONTEND_DIR.is_dir():
  1539→        return None
  1540→    root = _FRONTEND_DIR.resolve()
  1541→    try:
  1542→        candidate = (root / path).resolve()
  1543→        candidate.relative_to(root)
  1544→    except ValueError:
  1545→        return None
  1546→    if candidate.is_dir():
  1547→        index = candidate / "index.html"
  1548→        return index if index.is_file() else None
  1549→    return candidate if candidate.is_file() else None

  1551→def _serve_frontend_index(index_path: Path) -> Response:
  1552→    html = index_path.read_text(encoding="utf-8")
  1553→    script = _frontend_config_script()
  1554→    if "</head>" in html:
  1555→        html = html.replace("</head>", f"{script}</head>", 1)
  1556→    else:
  1557→        html = f"{script}{html}"
  1558→    return Response(html, media_type="text/html")

  1560→def _serve_frontend_path(path: str):
  1561→    normalized = path.strip("/")
  1562→    if normalized and _is_reserved_frontend_path(normalized):
  1563→        return JSONResponse({"detail": "Not found"}, status_code=404)
  1564→
  1565→    file_path = _safe_frontend_file(normalized or "index.html")
  1566→    if file_path:
  1567→        if file_path.name == "index.html":
  1568→            return _serve_frontend_index(file_path)
  1569→        return FileResponse(file_path)
  1570→
  1571→    index_path = _safe_frontend_file("index.html")
  1572→    if index_path:
  1573→        return _serve_frontend_index(index_path)
  1574→    return _frontend_not_found()
